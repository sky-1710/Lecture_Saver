# Lecture Saver - runtime dependencies
#
# Everything here runs locally. No dependency in this file talks to a cloud
# transcription or vision service. The only network access the application ever
# performs is the ONE-TIME download of Whisper model weights from Hugging Face
# (via huggingface_hub, pulled in by faster-whisper). After that download the
# application works fully offline.

# --- Desktop UI -------------------------------------------------------------
# Essentials = QtCore/QtGui/QtWidgets. Addons is required as well because
# QtMultimedia + QtMultimediaWidgets moved into the Addons wheel in Qt 6.9.
# Qt's bundled FFmpeg media backend is what lets MP4/MOV/MKV play without the
# user installing any system codec pack.
PySide6-Essentials>=6.8,<6.10
PySide6-Addons>=6.8,<6.10

# --- Local speech-to-text ---------------------------------------------------
# CTranslate2 re-implementation of Whisper. Roughly 4x faster than
# openai-whisper on CPU with int8 quantisation, which matters a great deal on a
# machine with no CUDA GPU.
faster-whisper>=1.1.0

# Intel's inference runtime. Runs the Turbo model (Whisper large-v3-turbo) on an
# Intel integrated GPU - which CTranslate2 cannot use - about four times faster
# than faster-whisper's medium on the CPU. Local only; its model is a one-time
# Hugging Face download like the others. If it is missing, Turbo falls back to
# faster-whisper.
openvino-genai>=2026.4,<2027

# --- Reading slide text (keywords "From slides") ------------------------------
# Python bindings for the OCR engine built into Windows (Windows.Media.Ocr):
# part of the operating system, local and offline, no model to download. On
# other systems the button is shown as unavailable.
winrt-runtime>=3.2,<4; sys_platform == "win32"
winrt-Windows.Media.Ocr>=3.2,<4; sys_platform == "win32"
winrt-Windows.Graphics.Imaging>=3.2,<4; sys_platform == "win32"
winrt-Windows.Storage.Streams>=3.2,<4; sys_platform == "win32"
winrt-Windows.Foundation>=3.2,<4; sys_platform == "win32"
winrt-Windows.Foundation.Collections>=3.2,<4; sys_platform == "win32"
winrt-Windows.Globalization>=3.2,<4; sys_platform == "win32"
# Pillow (used to read slide images) already comes with reportlab below.

# --- Slide detection --------------------------------------------------------
# All frame comparison (perceptual hashing, per-cell SSIM, grid change maps) is
# plain NumPy. OpenCV is deliberately not used: FFmpeg already does the
# decoding, scaling and grayscale conversion, so OpenCV would add ~90 MB of
# wheel for operations that are a handful of array expressions.
numpy>=1.24

# --- FFmpeg binary ----------------------------------------------------------
# Ships a static ffmpeg.exe inside the wheel so FFmpeg does not have to be
# installed by hand. A system-wide ffmpeg/ffprobe is preferred when present.
imageio-ffmpeg>=0.5.1

# --- PDF export ---------------------------------------------------------------
# Lecture notes PDF built locally with ReportLab's layout engine (no browser, no
# service). Korean text is drawn with a TrueType font already on the machine
# (Malgun Gothic on Windows), embedded into the PDF. Pillow reads slide images.
reportlab>=4.2,<5
